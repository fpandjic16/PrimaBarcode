    LOCAL PROCEDURE UpdateRcptAmounts@50000();
    VAR
      WhseRcptLine@50000 : Record 7317;
      PurchLine@50001 : Record 39;
    BEGIN
      // <ADHR.058871>
      WhseRcptLine.SETCURRENTKEY("No.","Source Document");
      WhseRcptLine.SETRANGE("No.","No.");
      WhseRcptLine.SETRANGE("Source Document",WhseRcptLine."Source Document"::"Purchase Order");
      IF NOT WhseRcptLine.FINDSET THEN
        EXIT;

      REPEAT
        PurchLine.SETRANGE("Document No.",WhseRcptLine."Source No.");
        PurchLine.SETRANGE("Line No.",WhseRcptLine."Source Line No.");
        IF PurchLine.FINDFIRST THEN BEGIN
          WhseRcptLine.VALIDATE("Direct Unit Cost",PurchLine.Amount / PurchLine.Quantity);
          //<ZGNVSUPPCO-122>
          WhseRcptLine.VALIDATE("Line Discount %",PurchLine."Line Discount %");
          WhseRcptLine.VALIDATE("Unit Cost",PurchLine."Direct Unit Cost");
          //</ZGNVSUPPCO-122>
          WhseRcptLine.MODIFY;
        END;
      UNTIL WhseRcptLine.NEXT = 0;

      CurrPage.UPDATE(TRUE);
      // </ADHR.058871>
    END;
