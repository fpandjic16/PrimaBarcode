    BEGIN
      {

      }
    END;

      ERROR('UpdateStatusAfterShipping - redizajn'); // <PRICO-2025-01/>
      IF NOT IsCompletelyShipped_OLD() THEN
        EXIT;

      {
      IF "Receiving Location" = '' THEN
        VALIDATE(Status,Status::Shipped)
      ELSE BEGIN
        IF Location.GET("Receiving Location") THEN BEGIN
          IF Location."Require Receive" THEN
            VALIDATE(Status,Status::"Shipped - Awaiting Receipt")
          ELSE
            VALIDATE(Status,Status::Shipped);

        END ELSE
          VALIDATE(Status,Status::Shipped);
      END;
      }

      Location.SETRANGE(Code,"Destination No.");
      IF Location.ISEMPTY THEN
        VALIDATE(Status,Status::Shipped)
      ELSE
        VALIDATE(Status,Status::"Shipped - Awaiting Receipt");
    END;

    LOCAL PROCEDURE UpdateTransportQtyToReceiveOnTransferLines@50079(TransportDocNo@50000 : Code[20]);
    VAR
      TransportSheetLine@50001 : Record 50016;
      TransferLine@50003 : Record 5741;
      SourceNo@50005 : Code[20];
      TransferHeader@50002 : Record 5740;
      PostedWhseShptLine@50004 : Record 7323;
      DoModifyTransferHdr@50007 : Boolean;
      TempTransHdr@50008 : TEMPORARY Record 5740;
      TransportSheetLine2@50009 : Record 50016;
      TotalTransQtyOnCurrentTransportSheet@50010 : Decimal;
      DoModifyTransLine@50011 : Boolean;
      TransportSheetHeader@50012 : Record 50015;
    BEGIN
      TransportSheetHeader.GET(TransportDocNo);

      TransportSheetLine.SETRANGE("Document No.",TransportSheetHeader."Document No.");
      TransportSheetLine.SETRANGE("Source Type",TransportSheetLine."Source Type"::"Transfer Order");
      TransportSheetLine.SETFILTER("Qty. to Receive",'<>%1',0); // TODO
      IF TransportSheetLine.FINDSET THEN REPEAT
        //v2
        IF NOT TempTransHdr.GET(TransportSheetLine."Source No.") THEN BEGIN
          TempTransHdr.INIT;
          TempTransHdr."No." := TransportSheetLine."Source No.";
          TempTransHdr.INSERT;
        END;
        //v2

        DoModifyTransferHdr := FALSE;
        DoModifyTransLine := FALSE;

        IF NOT TransferLine.GET(TransportSheetLine."Source No.",TransportSheetLine."Source Line No.") THEN BEGIN
          IF TransportSheetLine."Qty. to Receive" <> 0 THEN
            TempErrorMessage.LogSimpleMessage(TempErrorMessage."Message Type"::Error,STRSUBSTNO(InvalidQtyToReceive,TransportSheetLine."Document No.",TransportSheetLine."Line No."));
        END ELSE BEGIN
          IF TransportSheetLine."Qty. Shipped" < TransportSheetLine."Qty. to Receive" THEN
            TempErrorMessage.LogSimpleMessage(TempErrorMessage."Message Type"::Error,STRSUBSTNO(InvalidQtyToReceive,TransportSheetLine."Document No.",TransportSheetLine."Line No."))
          ELSE BEGIN
            IF TransferLine."Qty. in Transit" < TransportSheetLine."Qty. to Receive" THEN
              TempErrorMessage.LogSimpleMessage(TempErrorMessage."Message Type"::Error,STRSUBSTNO(InvalidQtyToReceive,TransportSheetLine."Document No.",TransportSheetLine."Line No."))
            ELSE BEGIN
              PostedWhseShptLine.SETRANGE("Whse. Shipment No.",TransportSheetLine."Whse. Shipment No.");
              PostedWhseShptLine.SETRANGE("Whse Shipment Line No.",TransportSheetLine."Whse. Shipment Line No.");
              PostedWhseShptLine.FINDLAST;
              IF PostedWhseShptLine.Quantity <> TransportSheetLine."Qty. to Receive" THEN
                IF PostedWhseShptLine.Quantity < TransportSheetLine."Qty. to Receive" THEN
                  TempErrorMessage.LogSimpleMessage(TempErrorMessage."Message Type"::Error,STRSUBSTNO(InvalidQtyToReceive,TransportSheetLine."Document No.",TransportSheetLine."Line No."));

              IF TransferLine."Qty. to Receive" <> TransportSheetLine."Qty. to Receive" THEN
                IF TransferLine."Qty. in Transit" < TransportSheetLine."Qty. to Receive" THEN
                  TempErrorMessage.LogSimpleMessage(TempErrorMessage."Message Type"::Error,STRSUBSTNO(InvalidQtyToReceive,TransportSheetLine."Document No.",TransportSheetLine."Line No."))
                ELSE BEGIN
                  //v2
                  //get total transport line qty.
                  TotalTransQtyOnCurrentTransportSheet := TransportSheetLine."Qty. to Receive";

                  TransportSheetLine2.SETRANGE("Document No.",TransportSheetLine."Document No.");
                  TransportSheetLine2.SETRANGE("Source Type",TransportSheetLine2."Source Type"::"Transfer Order");
                  TransportSheetLine2.SETRANGE("Source No.",TransportSheetLine."Source No.");
                  TransportSheetLine2.SETRANGE("Source Line No.",TransportSheetLine."Source Line No.");
                  TransportSheetLine2.SETFILTER("Line No.",'<>%1',TransportSheetLine."Line No.");
                  IF TransportSheetLine2.FINDSET THEN BEGIN
                    REPEAT
                      TotalTransQtyOnCurrentTransportSheet += TransportSheetLine2."Qty. to Receive";
                    UNTIL TransportSheetLine2.NEXT = 0;
                  END;

                  IF TransferLine."Qty. to Receive" <> TotalTransQtyOnCurrentTransportSheet THEN BEGIN
                    TransferLine.VALIDATE(TransferLine."Qty. to Receive",TotalTransQtyOnCurrentTransportSheet);
                    DoModifyTransLine := TRUE;
                  END;
                  //v2
                END;
            END;
          END;

          //v2
          IF TransferLine."Transport Sheet No." <> TransportSheetHeader."Document No." THEN BEGIN
            TransferLine."Transport Sheet No." := TransportSheetHeader."Document No.";
            DoModifyTransLine := TRUE;
          END;

          IF DoModifyTransLine THEN
            TransferLine.MODIFY;
          //v2

          IF (SourceNo <> TransportSheetLine."Source No.") THEN
            IF NOT TempErrorMessage.HasErrors(FALSE) THEN BEGIN
              TransferHeader.GET(TransportSheetLine."Source No.");

              TransferHeader.SetHideValidationDialog(TRUE);
              IF NOT TransferHeader."Transport Sheet Posting" THEN BEGIN
                TransferHeader.VALIDATE("Transport Sheet Posting",TRUE);
                DoModifyTransferHdr := TRUE;
              END;

              IF TransferHeader."Posting Date" <> WORKDATE THEN BEGIN
                TransferHeader.VALIDATE("Posting Date", WORKDATE);
                DoModifyTransferHdr := TRUE;
              END;

              IF DoModifyTransferHdr THEN
                TransferHeader.MODIFY;

              TransferHeader.SetHideValidationDialog(FALSE);
            END;

          SourceNo := TransportSheetLine."Source No.";
        END;
      UNTIL TransportSheetLine.NEXT = 0
      ELSE
        TempErrorMessage.LogSimpleMessage(TempErrorMessage."Message Type"::Error,STRSUBSTNO(NoTransportSheetLineToReceive,TransportDocNo));

      //v2
      TempTransHdr.RESET;
      TransportSheetLine.RESET;
      IF TempTransHdr.FINDSET THEN REPEAT
        TransferLine.RESET;

    LOCAL PROCEDURE UpdateTransportSheetQtyToReceive@50015(TransferLine@50001 : Record 5741;TransportSheetLine@50000 : Record 50016);
    VAR
      DoModifyTransferHdr@50006 : Boolean;
      TransportSheetLine2@50005 : Record 50016;
      TotalTransQtyOnCurrentTransportSheet@50004 : Decimal;
      DoModifyTransLine@50003 : Boolean;
      TransportSheetHeader2@50002 : Record 50015;
      PostedWhseShptLine@50007 : Record 7323;
      SourceNo@50008 : Code[20];
      TransferHeader@50009 : Record 5740;
    BEGIN
      //v2
      DoModifyTransferHdr := FALSE;
      DoModifyTransLine := FALSE;

      IF NOT TransferLine.GET(TransportSheetLine."Source No.",TransportSheetLine."Source Line No.") THEN BEGIN
        IF TransportSheetLine."Qty. to Receive" <> 0 THEN
          TempErrorMessage.LogSimpleMessage(TempErrorMessage."Message Type"::Error,STRSUBSTNO(InvalidQtyToReceive,TransportSheetLine."Document No.",TransportSheetLine."Line No."));
      END ELSE BEGIN
        IF TransportSheetLine."Qty. to Receive" = 0 THEN BEGIN
          IF TransferLine."Qty. to Receive" <> 0 THEN BEGIN
            TransferLine.VALIDATE("Qty. to Receive", 0);
            DoModifyTransLine := TRUE;
          END;
        END ELSE IF TransportSheetLine."Qty. Shipped" < TransportSheetLine."Qty. to Receive" THEN
          TempErrorMessage.LogSimpleMessage(TempErrorMessage."Message Type"::Error,STRSUBSTNO(InvalidQtyToReceive,TransportSheetLine."Document No.",TransportSheetLine."Line No."))
        ELSE BEGIN
          IF TransferLine."Qty. in Transit" < TransportSheetLine."Qty. to Receive" THEN
            TempErrorMessage.LogSimpleMessage(TempErrorMessage."Message Type"::Error,STRSUBSTNO(InvalidQtyToReceive,TransportSheetLine."Document No.",TransportSheetLine."Line No."))
          ELSE BEGIN
            PostedWhseShptLine.SETRANGE("Whse. Shipment No.",TransportSheetLine."Whse. Shipment No.");
            PostedWhseShptLine.SETRANGE("Whse Shipment Line No.",TransportSheetLine."Whse. Shipment Line No.");
            PostedWhseShptLine.FINDLAST;
            IF PostedWhseShptLine.Quantity <> TransportSheetLine."Qty. to Receive" THEN
              IF PostedWhseShptLine.Quantity < TransportSheetLine."Qty. to Receive" THEN
                TempErrorMessage.LogSimpleMessage(TempErrorMessage."Message Type"::Error,STRSUBSTNO(InvalidQtyToReceive,TransportSheetLine."Document No.",TransportSheetLine."Line No."));

            IF TransferLine."Qty. to Receive" <> TransportSheetLine."Qty. to Receive" THEN BEGIN
              //get total transport line qty.
              TotalTransQtyOnCurrentTransportSheet := TransportSheetLine."Qty. to Receive";

              TransportSheetLine2.SETRANGE("Document No.",TransportSheetLine."Document No.");
              TransportSheetLine2.SETRANGE("Source Type",TransportSheetLine2."Source Type"::"Transfer Order");
              TransportSheetLine2.SETRANGE("Source No.",TransportSheetLine."Source No.");
              TransportSheetLine2.SETRANGE("Source Line No.",TransportSheetLine."Source Line No.");
              TransportSheetLine2.SETFILTER("Line No.",'<>%1',TransportSheetLine."Line No.");
              IF TransportSheetLine2.FINDSET THEN BEGIN
                REPEAT
                  TotalTransQtyOnCurrentTransportSheet += TransportSheetLine2."Qty. to Receive";
                UNTIL TransportSheetLine2.NEXT = 0;
              END;

              IF TransferLine."Qty. to Receive" <> TotalTransQtyOnCurrentTransportSheet THEN BEGIN
                TransferLine.VALIDATE(TransferLine."Qty. to Receive",TotalTransQtyOnCurrentTransportSheet);
                DoModifyTransLine := TRUE;
              END;
            END;
          END;
        END;

        IF TransferLine."Transport Sheet No." <> TransportSheetLine."Document No." THEN BEGIN
          TransferLine."Transport Sheet No." := TransportSheetLine."Document No.";
          DoModifyTransLine := TRUE;
        END;

        IF DoModifyTransLine THEN
          TransferLine.MODIFY;
      END;
      //v2
    END;
