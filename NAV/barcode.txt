    [EventSubscriber(Table,5401813,OnAfterModifyEvent)]
    LOCAL PROCEDURE BEMFRecording_OnAfterModify@50003(VAR Rec@50000 : Record 5401813;VAR xRec@50001 : Record 5401813;RunTrigger@50002 : Boolean);
    VAR
      BEMFRecording@50003 : Record 5401813;
      DocNo@50004 : Code[20];
      ScannedDocumentType@50005 : 'Shipment,Transfer Order,Inventory,Prereceipt,Transport,Receipt,WHCDSHIP,WHCDREC,WHCDSHFROM,CPCDSHIP,CPCDREC,CPCDSHFROM';
    BEGIN
      //due to SOAP call, cannot use OnAfterInsert
      IF Rec.ISTEMPORARY THEN
        EXIT;

      IF Rec.orderId <> '' THEN
        DocNo := Rec.orderId
      ELSE
        DocNo := Rec.description;

      IF DocNo = '' THEN
        ERROR(STRSUBSTNO(DocumentNotExists, Rec));

      CASE Rec.orderType OF
        UPPERCASE(FORMAT(ScannedDocumentType::Shipment)) :
          CheckDocumentStatus(ScannedDocumentType::Shipment,DocNo);
        UPPERCASE(FORMAT(ScannedDocumentType::Prereceipt)) :
          CheckDocumentStatus(ScannedDocumentType::Prereceipt,DocNo);
        //<ZGNVSUPPCO-1059>
        UPPERCASE(FORMAT(ScannedDocumentType::Receipt)) :
          CheckDocumentStatus(ScannedDocumentType::Receipt,DocNo);
        //</ZGNVSUPPCO-1059>
        // <ZGNVSUPPCO-1246>
        UPPERCASE(FORMAT(ScannedDocumentType::Transport)) :
          CheckDocumentStatus(ScannedDocumentType::Transport,DocNo);
        // </ZGNVSUPPCO-1246>
        // <ZGNVSUPPCO-1246>
        UPPERCASE(FORMAT(ScannedDocumentType::WHCDSHIP)) :
          CheckDocumentStatus(ScannedDocumentType::WHCDSHIP,DocNo);
        UPPERCASE(FORMAT(ScannedDocumentType::WHCDREC)) :
          CheckDocumentStatus(ScannedDocumentType::WHCDREC,DocNo);
        UPPERCASE(FORMAT(ScannedDocumentType::WHCDSHFROM)) :
          CheckDocumentStatus(ScannedDocumentType::WHCDSHFROM,DocNo);
        UPPERCASE(FORMAT(ScannedDocumentType::CPCDSHIP)) :
          CheckDocumentStatus(ScannedDocumentType::CPCDSHIP,DocNo);
        UPPERCASE(FORMAT(ScannedDocumentType::CPCDREC)) :
          CheckDocumentStatus(ScannedDocumentType::CPCDREC,DocNo);
        UPPERCASE(FORMAT(ScannedDocumentType::CPCDSHFROM)) :
          CheckDocumentStatus(ScannedDocumentType::CPCDSHFROM,DocNo);
        // </ZGNVSUPPCO-1246>
      END;

      CreateSyncRecord(DocNo, Rec.orderType, Rec."User ID"); //<ZGNVSUPPCO-1059/>
    END;

    LOCAL PROCEDURE CheckDocumentStatus@50006(DocumentType@50000 : 'Shipment,Transfer Order,Inventory,Prereceipt,Transport,Receipt,WHCDSHIP,WHCDREC,WHCDSHFROM,CPCDSHIP,CPCDREC,CPCDSHFROM';DocumentNo@50001 : Code[20]);
    VAR
      WhseShpt@50002 : Record 7320;
      DocDoesntExists@50003 : TextConst 'ENU=%1 %2 does not exists or has unallowed status.;HRV=%1 %2 ne postoji ili je u nedozvoljenom statusu.';
      WhseRcpt@50004 : Record 7316;
      TransSheet@50005 : Record 50015;
      ComplaintShpt@50006 : Record 50021;
    BEGIN
      CASE DocumentType OF
        DocumentType::Shipment :
          BEGIN
            WhseShpt.LOCKTABLE(TRUE);
            IF NOT (WhseShpt.GET(DocumentNo)) OR (WhseShpt."Barcode Status" = WhseShpt."Barcode Status"::Processed) THEN
              ERROR(STRSUBSTNO(DocDoesntExists,WhseShpt.TABLECAPTION,DocumentNo))
            ELSE IF (WhseShpt."Barcode Status" = WhseShpt."Barcode Status"::Blank) THEN BEGIN
              WhseShpt."Barcode Status" := WhseShpt."Barcode Status"::Synchronized;
              WhseShpt.MODIFY;
            END;
          END;
        //<ZGNVSUPPCO-1059>
        DocumentType::Receipt:
          BEGIN
            WhseRcpt.LOCKTABLE(TRUE);
            IF NOT (WhseRcpt.GET(DocumentNo)) OR (WhseRcpt."Barcode Status" = WhseRcpt."Barcode Status"::Processed) THEN
              ERROR(STRSUBSTNO(DocDoesntExists,WhseRcpt.TABLECAPTION,DocumentNo));

            IF (WhseRcpt."Barcode Status" = WhseRcpt."Barcode Status"::Blank) THEN BEGIN
              WhseRcpt."Barcode Status" := WhseRcpt."Barcode Status"::Synchronized;
              WhseRcpt.MODIFY;
            END;
          END;
        //</ZGNVSUPPCO-1059>
        DocumentType::Transport:
        //<ZGNVSUPPCO-1246>
          BEGIN
            TransSheet.LOCKTABLE(TRUE);
            IF NOT (TransSheet.GET(DocumentNo)) OR (TransSheet."Barcode Status" = TransSheet."Barcode Status"::Processed) THEN
              ERROR(STRSUBSTNO(DocDoesntExists,TransSheet.TABLECAPTION,DocumentNo));

            IF (TransSheet."Barcode Status" = TransSheet."Barcode Status"::Blank) THEN BEGIN
              TransSheet."Barcode Status" := TransSheet."Barcode Status"::Synchronized;
              TransSheet.MODIFY;
            END;
          END;
        //</ZGNVSUPPCO-1246>
        //<ZGNVSUPPCO-1245>
        DocumentType::WHCDSHIP:
          BEGIN
            WhseShpt.LOCKTABLE(TRUE);
            IF NOT WhseShpt.GET(DocumentNo) OR (WhseShpt."Cross-Dock Shipment Status" > WhseShpt."Cross-Dock Shipment Status"::"Shipped to CD") THEN
              ERROR(STRSUBSTNO(DocDoesntExists,WhseShpt.TABLECAPTION,DocumentNo))
            ELSE IF (WhseShpt."Cross-Dock Shipment Status" < WhseShpt."Cross-Dock Shipment Status"::"Shipped to CD (sync)") THEN BEGIN
              WhseShpt."Cross-Dock Shipment Status" := WhseShpt."Cross-Dock Shipment Status"::"Shipped to CD (sync)";
              WhseShpt.MODIFY;
            END;
          END;
        DocumentType::WHCDREC:
          BEGIN
            WhseShpt.LOCKTABLE(TRUE);
            IF NOT WhseShpt.GET(DocumentNo) THEN
              ERROR(STRSUBSTNO(DocDoesntExists,WhseShpt.TABLECAPTION,DocumentNo))
            ELSE IF (WhseShpt."Cross-Dock Shipment Status" < WhseShpt."Cross-Dock Shipment Status"::"Shipped to CD (sync)") OR (WhseShpt."Cross-Dock Shipment Status" > WhseShpt."Cross-Dock Shipment Status"::"Received on CD") THEN
              ERROR(STRSUBSTNO(DocDoesntExists,WhseShpt.TABLECAPTION,DocumentNo))
            ELSE IF (WhseShpt."Cross-Dock Shipment Status" < WhseShpt."Cross-Dock Shipment Status"::"Received on CD (sync)") THEN BEGIN
              WhseShpt."Cross-Dock Shipment Status" := WhseShpt."Cross-Dock Shipment Status"::"Received on CD (sync)";
              WhseShpt.MODIFY;
            END;
          END;
        DocumentType::WHCDSHFROM:
          BEGIN
            WhseShpt.LOCKTABLE(TRUE);
            IF NOT WhseShpt.GET(DocumentNo) THEN
              ERROR(STRSUBSTNO(DocDoesntExists,WhseShpt.TABLECAPTION,DocumentNo))
            ELSE IF (WhseShpt."Cross-Dock Shipment Status" < WhseShpt."Cross-Dock Shipment Status"::"Received on CD (sync)") THEN
              ERROR(STRSUBSTNO(DocDoesntExists,WhseShpt.TABLECAPTION,DocumentNo))
            ELSE IF (WhseShpt."Cross-Dock Shipment Status" < WhseShpt."Cross-Dock Shipment Status"::"Received on CD (sync)") THEN BEGIN
              WhseShpt."Cross-Dock Shipment Status" := WhseShpt."Cross-Dock Shipment Status"::"Received on CD (sync)";
              WhseShpt.MODIFY;
            END;
          END;

        DocumentType::CPCDSHIP:
          BEGIN
            ComplaintShpt.LOCKTABLE(TRUE);
            IF NOT ComplaintShpt.GET(DocumentNo) OR (ComplaintShpt."Cross-Dock Shipment Status" > ComplaintShpt."Cross-Dock Shipment Status"::"Shipped to CD") THEN
              ERROR(STRSUBSTNO(DocDoesntExists,ComplaintShpt.TABLECAPTION,DocumentNo))
            ELSE IF (ComplaintShpt."Cross-Dock Shipment Status" < ComplaintShpt."Cross-Dock Shipment Status"::"Shipped to CD (sync)") THEN BEGIN
              ComplaintShpt."Cross-Dock Shipment Status" := ComplaintShpt."Cross-Dock Shipment Status"::"Shipped to CD (sync)";
              ComplaintShpt.MODIFY;
            END;
          END;
        DocumentType::CPCDREC:
          BEGIN
            ComplaintShpt.LOCKTABLE(TRUE);
            IF NOT ComplaintShpt.GET(DocumentNo) THEN
              ERROR(STRSUBSTNO(DocDoesntExists,ComplaintShpt.TABLECAPTION,DocumentNo))
            ELSE IF (ComplaintShpt."Cross-Dock Shipment Status" < ComplaintShpt."Cross-Dock Shipment Status"::"Shipped to CD (sync)") OR (ComplaintShpt."Cross-Dock Shipment Status" > ComplaintShpt."Cross-Dock Shipment Status"::"Received on CD") THEN
              ERROR(STRSUBSTNO(DocDoesntExists,ComplaintShpt.TABLECAPTION,DocumentNo))
            ELSE IF (ComplaintShpt."Cross-Dock Shipment Status" < ComplaintShpt."Cross-Dock Shipment Status"::"Received on CD (sync)") THEN BEGIN
              ComplaintShpt."Cross-Dock Shipment Status" := ComplaintShpt."Cross-Dock Shipment Status"::"Received on CD (sync)";
              ComplaintShpt.MODIFY;
            END;
          END;
        DocumentType::CPCDSHFROM:
          BEGIN
            ComplaintShpt.LOCKTABLE(TRUE);
            IF NOT ComplaintShpt.GET(DocumentNo) THEN
              ERROR(STRSUBSTNO(DocDoesntExists,ComplaintShpt.TABLECAPTION,DocumentNo))
            ELSE IF (ComplaintShpt."Cross-Dock Shipment Status" < ComplaintShpt."Cross-Dock Shipment Status"::"Received on CD (sync)") THEN
              ERROR(STRSUBSTNO(DocDoesntExists,ComplaintShpt.TABLECAPTION,DocumentNo))
            ELSE IF (ComplaintShpt."Cross-Dock Shipment Status" < ComplaintShpt."Cross-Dock Shipment Status"::"Received on CD (sync)") THEN BEGIN
              ComplaintShpt."Cross-Dock Shipment Status" := ComplaintShpt."Cross-Dock Shipment Status"::"Received on CD (sync)";
              ComplaintShpt.MODIFY;
            END;
          END;
        // </ZGNVSUPPCO-1245>
      END;
    END;

    PROCEDURE FillWhseShptBarcodeEntries@50004(Rec@50000 : Record 7320;SkipRelease@50006 : Boolean);
    VAR
      BEMFOrder@50003 : Record 5401810;
      BEMFOrderLine@50002 : Record 5401811;
      BEMFOrderLineItem@50001 : Record 5401812;
      WarehouseShipmentLine@50004 : Record 7321;
      BOMComponent@50005 : Record 90;
      UnitofMeasure@50007 : Record 204;
      BarcodeNo@50008 : Text[200];
      DocTypeLoc@50009 : 'Shipment,Transfer Order,Inventory,Prereceipt,Transport,Receipt,WHCDSHIP,WHCDREC,WHCDSHFROM,CPCDSHIP,CPCDREC,CPCDSHFROM';
      IsRetailLocation@50010 : Boolean;
    BEGIN
      CLEAR(LineNo);

      IF NOT BarcodeMgmt.IsBeMobileLocation(Rec."No.",DocTypeLoc::Shipment) THEN
        EXIT;

      IF Rec.Status = Rec.Status::Released THEN
        IF NOT SkipRelease THEN
          EXIT;

      IF Rec."Barcode Status" = Rec."Barcode Status"::Processed THEN
        EXIT;

      // <ZGNVSUPPCO-1245>
      Rec.CALCFIELDS("Retail Location");
      IsRetailLocation := Rec."Retail Location";
      //IF BEMFOrder.GET(Rec."No.") THEN
      IF BEMFOrder.GET(Rec."No.",FORMAT(DocTypeLoc::Shipment)) THEN
      // </ZGNVSUPPCO-1245>
        //DeleteWhseShptBarcodeEntries(Rec); // <ZGNVSUPPCO-1059/>
        DeleteBarcodeEntries(Rec."No.",DocTypeLoc::Shipment);

      WarehouseShipmentLine.RESET;
      WarehouseShipmentLine.SETRANGE("No.",Rec."No.");
      IF NOT WarehouseShipmentLine.FINDSET THEN
        EXIT;

      BEMFOrder.INIT;
      BEMFOrder.orderId := Rec."No.";
      BEMFOrder.orderType := FORMAT(DocTypeLoc::Shipment);
      BEMFOrder.description_1 := FORMAT(Rec."Posting Date")+', '+ Rec.FIELDCAPTION("Destination No.")+ ': ' +Rec."Destination No."+', '+ Rec.FIELDCAPTION("Location Code")+ ': ' +Rec."Location Code";
      BEMFOrder.description_2 := FORMAT(Rec."Location Code");
      BEMFOrder.description_4 := FORMAT(Rec."Posting Date");
      // <ZGNVSUPPCO-1246>
      //BEMFOrder.description_3 := FORMAT(CURRENTDATETIME);
      ResponsibilityCenterLocation.SETRANGE("Location Code",Rec."Location Code");
      IF ResponsibilityCenterLocation.FINDFIRST THEN
        BEMFOrder.description_3 := ResponsibilityCenterLocation."Responsibility Center";
      BEMFOrder.RetailLocation := IsRetailLocation;
      // </ZGNVSUPPCO-1246>

      BEMFOrder.INSERT;
      REPEAT
        BOMComponent.RESET;
        BOMComponent.SETRANGE("Parent Item No.",WarehouseShipmentLine."Item No.");
        BOMComponent.SETRANGE(Type,BOMComponent.Type::Item);
        IF BOMComponent.FINDSET THEN BEGIN
          REPEAT
            LineNo+=10000;
            BarcodeNo := GetItemBarcode(BOMComponent."No.",BOMComponent."Unit of Measure Code",BOMComponent."Variant Code");
            IF UnitofMeasure.GET(BOMComponent."Unit of Measure Code") AND UnitofMeasure."Insert Qty Barcode" THEN BEGIN
              BarcodeNo += '|' + BOMComponent."Unit of Measure Code" + '|' + FORMAT(BOMComponent."Quantity per");
              InsertSplitBEMFEntries(BEMFOrder,DocTypeLoc::Shipment,WarehouseShipmentLine."Line No.",TRUE,BOMComponent,BarcodeNo)
            END ELSE
              //InsertBEMFLineEntry(BEMFOrder,WarehouseShipmentLine,TRUE,BOMComponent, BarcodeNo); //<ZGNVSUPPCO-1059/>
              InsertBEMFLineEntry(BEMFOrder,DocTypeLoc::Shipment,WarehouseShipmentLine."Line No.",TRUE,BOMComponent, BarcodeNo); //<ZGNVSUPPCO-1059/>
          UNTIL BOMComponent.NEXT = 0;

        END ELSE BEGIN //normal items
          LineNo+=10000;
          BarcodeNo := GetItemBarcode(WarehouseShipmentLine."Item No.",WarehouseShipmentLine."Unit of Measure Code",WarehouseShipmentLine."Variant Code");
          IF UnitofMeasure.GET(WarehouseShipmentLine."Unit of Measure Code") AND UnitofMeasure."Insert Qty Barcode" THEN BEGIN
            // <ZGNVSUPPCO-1059>
            //BarcodeNo += '|' + WarehouseShipmentLine."Unit of Measure Code" + '|' + FORMAT(WarehouseShipmentLine.Quantity);
            BarcodeNo += '|' + WarehouseShipmentLine."Unit of Measure Code" + '|' + FORMAT(WarehouseShipmentLine."Qty. to Ship");
            // </ZGNVSUPPCO-1059>
            InsertSplitBEMFEntries(BEMFOrder,DocTypeLoc::Shipment,WarehouseShipmentLine."Line No.",FALSE,BOMComponent,BarcodeNo)
          END ELSE
            //InsertBEMFLineEntry(BEMFOrder,WarehouseShipmentLine,FALSE,BOMComponent,BarcodeNo);  //<ZGNVSUPPCO-1059/>
            InsertBEMFLineEntry(BEMFOrder,DocTypeLoc::Shipment,WarehouseShipmentLine."Line No.",FALSE,BOMComponent,BarcodeNo);  //<ZGNVSUPPCO-1059/>
        END;
      UNTIL WarehouseShipmentLine.NEXT = 0;
    END;

    PROCEDURE DeleteBarcodeEntries@50005(DocNo@50000 : Code[20];DocTypeLoc@50009 : 'Shipment,Transfer Order,Inventory,Prereceipt,Transport,Receipt,WHCDSHIP,WHCDREC,WHCDSHFROM,CPCDSHIP,CPCDREC,CPCDSHFROM');
    VAR
      BEMFOrder@50003 : Record 5401810;
      BEMFOrderLine@50002 : Record 5401811;
      BEMFOrderLineItem@50001 : Record 5401812;
      WhseShptFound@50004 : Boolean;
      WarehouseShipmentHeader@50006 : Record 7320;
      WarehouseReceiptHeader@50007 : Record 7316;
      FusionSetup@50008 : Record 5401802;
    BEGIN
      // <ZGNVSUPPCO-1245>
      {
      //<ZGNVSUPPCO-1059>
      //IF Rec.Status = Rec.Status::Open THEN
      //  EXIT;
      {CASE DocType OF
        DocType::Shipment:
          BEGIN
            WarehouseShipmentHeader.GET(DocNo);
            IF WarehouseShipmentHeader.Status = WarehouseShipmentHeader.Status::Open THEN
              EXIT;
          END;
        DocType::Receipt:
          BEGIN
            WarehouseReceiptHeader.GET(DocNo);
            IF WarehouseReceiptHeader.Status = WarehouseReceiptHeader.Status::Open THEN
              EXIT;
          END;
      END;}
      //</ZGNVSUPPCO-1059>

      // <ZGNVSUPPCO-1059>
      //IF BEMFOrder.GET(Rec."No.") THEN
      IF BEMFOrder.GET(DocNo) THEN
      // </ZGNVSUPPCO-1059>
        WhseShptFound := TRUE
      ELSE BEGIN
        //<ZGNVSUPPCO-1059>
        //BEMFOrder.SETRANGE(description_1,Rec."No.");
        //BEMFOrder.SETFILTER(orderType,FORMAT(ScannedDocumentType::Shipment));
        BEMFOrder.SETRANGE(description_1,DocNo);
        BEMFOrder.SETFILTER(orderType, '%1',FORMAT(DocType));
        //<ZGNVSUPPCO-1059>
        IF BEMFOrder.FINDFIRST THEN
          WhseShptFound := TRUE;
      END;
      }
      // <ZGNVSUPPCO-1245>
      {IF NOT FusionSetup.FINDFIRST THEN
        EXIT;

      BEMFOrder.SETRANGE(orderType,FusionSetup."Primary Key");
      BEMFOrder.SETRANGE(orderId,DocNo);
      BEMFOrder.SETRANGE(orderType,FORMAT(ScannedDocumentType::Shipment));
      }

      IF BEMFOrder.GET(DocNo, FORMAT(DocTypeLoc)) THEN
      // </ZGNVSUPPCO-1245>
        WhseShptFound := TRUE
      ELSE BEGIN
        BEMFOrder.SETRANGE(description_1,DocNo);
        BEMFOrder.SETRANGE(orderType,FORMAT(DocTypeLoc)); // <ZGNVSUPPCO-1245/>
        BEMFOrder.SETRANGE(orderId);
        IF BEMFOrder.FINDFIRST THEN
          WhseShptFound := TRUE;
      END;

      IF WhseShptFound THEN BEGIN
        BEMFOrderLineItem.SETRANGE(orderId,BEMFOrder.orderId);
        BEMFOrderLineItem.SETRANGE(orderType,FORMAT(DocTypeLoc)); // <ZGNVSUPPCO-1245/>

        IF BEMFOrderLineItem.FINDSET THEN
          BEMFOrderLineItem.DELETEALL;
        BEMFOrderLine.SETRANGE(orderId,BEMFOrder.orderId);
        BEMFOrderLine.SETRANGE(orderType,FORMAT(DocTypeLoc)); // <ZGNVSUPPCO-1245/>
        IF BEMFOrderLine.FINDSET THEN
          BEMFOrderLine.DELETEALL;
        BEMFOrder.DELETE;
      END;
    END;

    PROCEDURE DeleteScannedEntries@50002(DocNo@50001 : Code[20]);
    VAR
      ScannedDocumentLines@50002 : Record 50111;
    BEGIN
      // <ZGNVSUPPCO-1245>
      {
      CASE DocType OF
        ScannedDocumentType::Shipment: BEGIN
          WarehouseShipmentHeader.GET(DocNo);
          DeleteBarcodeEntries(WarehouseShipmentHeader."No.", DocType);
        END;
        //<ZGNVSUPPCO-1059>
        ScannedDocumentType::Receipt: BEGIN
          WarehouseReceiptHeader.GET(DocNo);
          DeleteBarcodeEntries(WarehouseReceiptHeader."No.", DocType);
        END;
        //</ZGNVSUPPCO-1059>
        ScannedDocumentType::Prereceipt:;
      END;
      }

      ScannedDocumentLines.SETRANGE("Document No.",DocNo);
      IF ScannedDocumentLines.FINDSET THEN
        ScannedDocumentLines.DELETEALL(FALSE);
      // </ZGNVSUPPCO-1245>
    END;

    LOCAL PROCEDURE GetItemBarcode@50010(ItemNo@50000 : Code[20];UomCode@50003 : Code[10];VariantCode@50004 : Code[10]) : Code[50];
    VAR
      Barcodes@50001 : Record 99001451;
      Item@50002 : Record 27;
    BEGIN
      IF (NOT Item.GET(ItemNo)) OR (Item.Type = Item.Type::Service) THEN
        EXIT;

    PROCEDURE GetBarcodeFields@50011(BEMFRecording@50001 : Record 5401813);
    VAR
      Barcodes@50002 : Record 99001451;
      Item@50003 : Record 27;
      BarcodeNo@50000 : Text[40];
    BEGIN
      CLEAR(BarcodeItem);
      CLEAR(BarcodeUoM);
      CLEAR(BarcodeQty);

        BarcodeItem := Barcodes."Item No.";

      END;
    END;

      END;

      BEMFRecording.SETRANGE(orderId,DocNo);

      IF EntriesInserted THEN
        DeleteBarcodeEntries(DocNo,DocType);
    END;
